cmake_minimum_required(VERSION 3.16)
project(mohdm6_imitation_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(imitation_tests test_policy.cpp ../../code/fgame/imitation_policy.cpp)
target_include_directories(imitation_tests PRIVATE ../../code/fgame)
if(MSVC)
    target_compile_options(imitation_tests PRIVATE /W4)
else()
    target_compile_options(imitation_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME imitation_core COMMAND imitation_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME imitation_data COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p test_data.py -v)
endif()
add_executable(imitation_runtime_tests test_runtime.cpp ../../code/fgame/imitation_policy.cpp)
target_include_directories(imitation_runtime_tests PRIVATE ../../code/fgame)
add_test(NAME imitation_runtime COMMAND imitation_runtime_tests)
if(Python3_FOUND)
    add_test(NAME imitation_runtime_log COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p test_runtime_log.py -v)
endif()

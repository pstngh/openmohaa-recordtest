cmake_minimum_required(VERSION 3.16)
project(spawn_replay_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(replay_tests test_replay.cpp ../../code/fgame/replay_track.cpp)
target_include_directories(replay_tests PRIVATE ../../code/fgame)
if(MSVC)
    target_compile_options(replay_tests PRIVATE /W4)
else()
    target_compile_options(replay_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME replay_core COMMAND replay_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME replay_importer COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p test_importer.py -v)
endif()

add_executable(replay_combat_tests test_replay_combat.cpp ../../code/fgame/replay_track.cpp)
target_include_directories(replay_combat_tests PRIVATE ../../code/fgame)
if(MSVC)
    target_compile_options(replay_combat_tests PRIVATE /W4)
else()
    target_compile_options(replay_combat_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME replay_live_combat COMMAND replay_combat_tests)
